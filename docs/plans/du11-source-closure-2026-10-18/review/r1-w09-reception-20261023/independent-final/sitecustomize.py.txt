reads = []
original_read = pathlib.Path.read_text
def read_text(self, *args, **kwargs):
    result = original_read(self, *args, **kwargs)
    if str(self.resolve()) in json.loads(os.environ.get("W09_WATCH", "[]")):
        reads.append({"path":str(self.resolve()),"returned_utf8_sha256":hashlib.sha256(result.encode("utf-8")).hexdigest(),"returned_utf8_bytes":len(result.encode("utf-8"))})
    return result
pathlib.Path.read_text = read_text
