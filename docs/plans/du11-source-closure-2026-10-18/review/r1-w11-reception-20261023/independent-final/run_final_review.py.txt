def frozen(paths):
    result = {}
    for relative in paths:
        data = (ROOT / relative).read_bytes()
        result[relative] = {"bytes": len(data), "sha256": hashlib.sha256(data).hexdigest(),
            "lf_sha256": hashlib.sha256(data.replace(b"\r\n", b"\n")).hexdigest()}
    return result
