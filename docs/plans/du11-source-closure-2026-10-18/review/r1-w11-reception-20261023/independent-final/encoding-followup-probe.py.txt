        os.environ.pop(name)

    raise RuntimeError("W11 independent encoding probe network blocked")

root = Path(__file__).resolve().parents[6]

from company_wiki.source_catalog import official_discovery as d
from company_wiki.document_normalization.html_parser import decode_html_bytes
assert d.decode_html_bytes is decode_html_bytes
source = Path(d.__file__)
assert hashlib.sha256(source.read_bytes()).hexdigest() == "6869349b5d6f4861e9c78cdc8748aaa7f547d2866cf278fc5f721418a137b123"
body = '<main><div>Loading…</div></main>'.encode("utf-8")
decoded, encoding, strategy, repaired = decode_html_bytes(body)
result = d.discover_official_documents(
    [{"url":"https://neutral.example/ir","original":body,
      "mime_type":"text/html","capture_id":"specific-short-utf8"}],
    entity="Neutral Holding",start_date="2026-01-01",as_of_date="2026-10-21")

print(json.dumps({"python":sys.version,"source":str(source),"shared_decoder_identity":True,
    "original_utf8_bytes":len(body),"original_utf8_sha256":hashlib.sha256(body).hexdigest(),
    "decoder":{"encoding":encoding,"strategy":strategy,"repaired":repaired},
    "actual_result":result},ensure_ascii=False))
